import { Request, Response, NextFunction } from 'express';
import jwt from 'jsonwebtoken';
import { JwtPayload, UserRole } from '../types';

export interface AuthRequest extends Request {
  user?: JwtPayload;
}

export function authenticate(req: AuthRequest, res: Response, next: NextFunction): void {
  const authHeader = req.headers.authorization;
  if (!authHeader?.startsWith('Bearer ')) {
    res.status(401).json({ error: 'No token provided' }); return;
  }
  const token = authHeader.substring(7);
  try {
    const decoded = jwt.verify(token, process.env.JWT_SECRET || 'secret') as JwtPayload;
    req.user = decoded;
    next();
  } catch {
    res.status(401).json({ error: 'Invalid or expired token' });
  }
}

export function authorize(...roles: UserRole[]) {
  return (req: AuthRequest, res: Response, next: NextFunction): void => {
    if (!req.user) { res.status(401).json({ error: 'Not authenticated' }); return; }
    if (!roles.includes(req.user.role)) { res.status(403).json({ error: 'Insufficient permissions' }); return; }
    next();
  };
}

export interface ValidationError {
  row: number;
  field: string;
  message: string;
}

export function validateCandidateRow(row: any, rowIndex: number): ValidationError[] {
  const errors: ValidationError[] = [];
  const r = rowIndex + 2; // 1-indexed, +1 for header

  if (!row.name || String(row.name).trim() === '') {
    errors.push({ row: r, field: 'name', message: 'Candidate name is required' });
  }
  if (!row.fatherName || String(row.fatherName).trim() === '') {
    errors.push({ row: r, field: 'fatherName', message: 'Father name is required' });
  }
  if (!row.phone || !/^\d{10}$/.test(String(row.phone).trim())) {
    errors.push({ row: r, field: 'phone', message: 'Phone must be exactly 10 digits' });
  }
  if (!row.dob) {
    errors.push({ row: r, field: 'dob', message: 'Date of birth is required' });
  } else {
    const dob = new Date(row.dob);
    if (isNaN(dob.getTime())) {
      errors.push({ row: r, field: 'dob', message: 'Invalid date of birth format' });
    } else {
      const age = (Date.now() - dob.getTime()) / (1000 * 60 * 60 * 24 * 365.25);
      if (age < 14) errors.push({ row: r, field: 'dob', message: 'Candidate must be at least 14 years old' });
    }
  }
  if (!row.courseCompleted || String(row.courseCompleted).trim() === '') {
    errors.push({ row: r, field: 'courseCompleted', message: 'Course completed is required' });
  }
  if (!row.completionDate) {
    errors.push({ row: r, field: 'completionDate', message: 'Completion date is required' });
  } else {
    const cd = new Date(row.completionDate);
    if (isNaN(cd.getTime())) {
      errors.push({ row: r, field: 'completionDate', message: 'Invalid completion date format' });
    } else if (cd > new Date()) {
      errors.push({ row: r, field: 'completionDate', message: 'Completion date cannot be in the future' });
    }
  }
  return errors;
}

export async function uploadCandidates(req: AuthRequest, res: Response): Promise<void> {
  try {
    const batchId = req.params.id;
    const batch = await prisma.batch.findUnique({ where: { id: batchId } });
    if (!batch) { res.status(404).json({ error: 'Batch not found' }); return; }
    if (!['DRAFT', 'RETURNED'].includes(batch.status)) {
      res.status(400).json({ error: 'Can only upload to DRAFT or RETURNED batches' }); return;
    }
    if (!req.file) { res.status(400).json({ error: 'No file uploaded' }); return; }

    const workbook = XLSX.readFile(req.file.path);
    const sheet = workbook.Sheets[workbook.SheetNames[0]];
    const rows: any[] = XLSX.utils.sheet_to_json(sheet, { defval: '' });

    if (rows.length === 0) {
      res.status(422).json({ error: 'File is empty or has no data rows' }); return;
    }

    // Validate ALL rows first
    const allErrors: ValidationError[] = [];
    const phones = new Set<string>();
    const aadhars = new Set<string>();

    rows.forEach((row, idx) => {
      const rowErrors = validateCandidateRow(row, idx);
      allErrors.push(...rowErrors);
      // Check duplicates within file
      const phone = String(row.phone || '').trim();
      if (phone && phones.has(phone)) {
        allErrors.push({ row: idx + 2, field: 'phone', message: `Duplicate phone number ${phone} in file` });
      }
      if (phone) phones.add(phone);
      const aadhar = String(row.aadhar || '').trim();
      if (aadhar && aadhars.has(aadhar)) {
        allErrors.push({ row: idx + 2, field: 'aadhar', message: `Duplicate aadhar ${aadhar} in file` });
      }
      if (aadhar) aadhars.add(aadhar);
    });

    // Clean up uploaded file
    fs.unlinkSync(req.file.path);

      return;
    }

    // All valid - save in a transaction
    await prisma.$transaction(async (tx) => {
      // Delete existing candidates for this batch (for resubmission)
      await tx.candidate.deleteMany({ where: { batchId } });
      // Insert all candidates
      await tx.candidate.createMany({
        data: rows.map(row => ({
          batchId,
          name: String(row.name).trim(),
          fatherName: String(row.fatherName).trim(),
          dob: new Date(row.dob),

          marks: row.marks ? parseFloat(row.marks) : null,
          grade: row.grade ? String(row.grade).trim() : null,
          status: 'PENDING'
        }))
      });
    });

    await createAuditLog({
      userId: req.user!.userId, action: 'UPLOAD_CANDIDATES', entityType: 'BATCH',

export async function approveBatch(req: AuthRequest, res: Response): Promise<void> {
  const { id: batchId } = req.params;
  const { remarks } = req.body;
  const officerId = req.user!.userId;

  try {
    // Use a transaction with serializable isolation to prevent race conditions
    const result = await prisma.$transaction(async (tx) => {
      // Lock the batch row
      const batch = await tx.batch.findUnique({
        where: { id: batchId },
        include: { _count: { select: { candidates: true } }, districtApprovals: true }
      });

      if (!batch) throw new Error('BATCH_NOT_FOUND');
      if (!['SUBMITTED', 'UNDER_REVIEW'].includes(batch.status)) throw new Error('INVALID_STATUS');

      // Check if this officer already approved
      const existing = batch.districtApprovals.find(a => a.officerId === officerId);
      if (existing) throw new Error('ALREADY_APPROVED');

      // Create approval record
      await tx.districtApproval.create({
        data: { batchId, officerId, decision: 'APPROVED', remarks: remarks || null }
      });

      // Count total approvals (including this new one)
      const approvalCount = batch.districtApprovals.filter(a => a.decision === 'APPROVED').length + 1;

      let newStatus = batch.status;
      if (approvalCount >= 2) {
        // Two approvals reached - advance
        const candidateCount = batch._count.candidates;
        newStatus = candidateCount >= QUALITY_THRESHOLD ? 'QUALITY_REVIEW' : 'STATE_REVIEW';
      } else {
        newStatus = 'UNDER_REVIEW';
      }

      const updated = await tx.batch.update({
        where: { id: batchId },
        data: { status: newStatus as any }
      });
      return { batch: updated, approvalCount, newStatus };
    });

    await createAuditLog({
      userId: officerId, action: 'APPROVE_BATCH', entityType: 'BATCH', entityId: batchId,
      batchId, previousStatus: 'UNDER_REVIEW', newStatus: result.newStatus,
      remarks: `District approval #${result.approvalCount}`
    });

    res.json({ message: 'Batch approved', batch: result.batch, approvalCount: result.approvalCount });
  } catch (err: any) {
    if (err.message === 'BATCH_NOT_FOUND') { res.status(404).json({ error: 'Batch not found' }); return; }
    if (err.message === 'INVALID_STATUS') { res.status(400).json({ error: 'Batch is not awaiting district approval' }); return; }
    if (err.message === 'ALREADY_APPROVED') { res.status(409).json({ error: 'You have already approved this batch' }); return; }
    console.error(err);
    res.status(500).json({ error: 'Failed to approve batch' });
  }
}

export async function generateCertificatesForBatch(batchId: string, issuedByUserId: string): Promise<void> {
  const batch = await prisma.batch.findUnique({

      candidates: true,

  if (!batch) return;

  const certDir = path.join(process.cwd(), 'certificates');
  if (!fs.existsSync(certDir)) fs.mkdirSync(certDir, { recursive: true });

  for (const candidate of batch.candidates) {
    // Idempotency: skip if certificate already exists
    const existing = await prisma.certificate.findUnique({ where: { candidateId: candidate.id } });
    if (existing) continue;

    const certNumber = generateCertNumber();
    const certId = uuidv4();
    const verifyUrl = `${process.env.FRONTEND_URL || 'http://localhost:5173'}/verify/${certNumber}`;

    // Generate QR code

    await new Promise<void>((resolve, reject) => {
      const doc = new PDFDocument({ size: 'A4', layout: 'landscape' });
      const stream = fs.createWriteStream(pdfPath);
      doc.pipe(stream);

      // Background
      doc.rect(0, 0, doc.page.width, doc.page.height).fill('#f0f8ff');
      
      // Border
      doc.rect(20, 20, doc.page.width - 40, doc.page.height - 40)

         .text('This is to certify that', 0, 120, { align: 'center' });

      // Candidate name
      doc.fillColor('#111827').fontSize(28).font('Helvetica-Bold')
         .text(candidate.name, 0, 150, { align: 'center' });

      doc.fillColor('#374151').fontSize(14).font('Helvetica')
         .text(`S/o D/o: ${candidate.fatherName}`, 0, 195, { align: 'center' });

      doc.fillColor('#1e40af').fontSize(20).font('Helvetica-Bold')
         .text(candidate.courseCompleted, 0, 250, { align: 'center' });

async function sendWhatsAppNotification(candidate: any, certificate: any): Promise<void> {
  // Check for existing notification (idempotency)
  const existing = await prisma.notification.findUnique({
    where: { candidateId_channel: { candidateId: candidate.id, channel: 'WHATSAPP' } }
  });
  if (existing?.status === 'SENT') return;

  let notif = existing;
  if (!notif) {
    notif = await prisma.notification.create({
      data: { candidateId: candidate.id, certificateId: certificate.id, channel: 'WHATSAPP', status: 'PENDING', attempts: 0 }
    });
  }

  const maxRetries = 3;
  for (let attempt = 1; attempt <= maxRetries; attempt++) {
    try {
      await prisma.notification.update({
        where: { id: notif!.id },
        data: { status: 'RETRYING', attempts: attempt, lastAttempt: new Date() }
      });
